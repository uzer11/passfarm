""" Оракул Звёзд — официальный бот (Bot API, aiogram 3.x). Личка: /start, /subscribe, /unsubscribe. Группы: /enable и /disable (только админ чата), /oracle (по запросу, с кулдауном). Админ: /broadcast <текст> — рассылка ТОЛЬКО подписавшимся. Запуск: BOT_TOKEN=... BOT_USERNAME=... ADMIN_ID=... python oracle_bot.py """ import asyncio import json import logging import os import time from pathlib import Path from aiogram import Bot, Dispatcher, F, Router from aiogram.client.default import DefaultBotProperties from aiogram.enums import ParseMode from aiogram.exceptions import TelegramForbiddenError, TelegramRetryAfter from aiogram.filters import Command, CommandStart from aiogram.types import (InlineKeyboardButton, InlineKeyboardMarkup, Message, WebAppInfo) BOT_TOKEN = os.environ["BOT_TOKEN"] BOT_USERNAME = os.environ.get("BOT_USERNAME", "") # без @, нужен для кнопки в группах ADMIN_ID = int(os.environ.get("ADMIN_ID", "0")) WEBAPP_URL = "https://uzer11.github.io/passfarm/" DB_PATH = Path("oracle_bot_db.json") ORACLE_COOLDOWN_SEC = 45 # не чаще одного ответа на чат в 45 сек BROADCAST_PACE_SEC = 0.05 # ~20 сообщений/сек, безопасно под лимиты router = Router() _last_oracle: dict[int, float] = {} def load_db() -> dict: if DB_PATH.exists(): try: return json.loads(DB_PATH.read_text(encoding="utf-8")) except Exception: pass return {"groups": [], "subs": []} def save_db(db: dict) -> None: DB_PATH.write_text(json.dumps(db, ensure_ascii=False), encoding="utf-8") def private_kb() -> InlineKeyboardMarkup: # web_app-кнопка работает в личке с любым https-URL return InlineKeyboardMarkup(inline_keyboard=[[ InlineKeyboardButton(text="🔮 Рассчитать матрицу судьбы", web_app=WebAppInfo(url=WEBAPP_URL)) ]]) def group_kb() -> InlineKeyboardMarkup: # В группах web_app-кнопки не работают — используем прямую ссылку на Mini App. # Настройте Main Mini App в BotFather: /mybots -> Bot Settings -> Configure Mini App. url = f"https://t.me/{BOT_USERNAME}/app" if BOT_USERNAME else WEBAPP_URL return InlineKeyboardMarkup(inline_keyboard=[[ InlineKeyboardButton(text="🔮 Открыть Оракул Звёзд", url=url) ]]) WELCOME = ( "✨ Оракул Звёзд\n\n" "По дате рождения рассчитаю ваш аркан и код судьбы — " "прогноз на неделю, месяц и год.\n\n" "Нажмите кнопку ниже, чтобы открыть интерактивный оракул.\n" "Развлекательный контент, 18+.\n\n" "/subscribe — ежедневный прогноз дня\n" "/unsubscribe — отписаться" ) ORACLE_TEXT = ( "🔮 Оракул Звёзд\n" "Интерактивный расчёт аркана по дате рождения: неделя, месяц, год.\n" "Развлекательный контент, 18+." ) @router.message(CommandStart(), F.chat.type == "private") async def cmd_start(message: Message): await message.answer(WELCOME, reply_markup=private_kb()) @router.message(Command("subscribe"), F.chat.type == "private") async def cmd_subscribe(message: Message): db = load_db() uid = message.from_user.id if uid not in db["subs"]: db["subs"].append(uid) save_db(db) await message.answer("✅ Подписка оформлена. Отписаться: /unsubscribe") @router.message(Command("unsubscribe"), F.chat.type == "private") async def cmd_unsubscribe(message: Message): db = load_db() uid = message.from_user.id if uid in db["subs"]: db["subs"].remove(uid) save_db(db) await message.answer("Вы отписались. Вернуться: /subscribe") async def is_chat_admin(bot: Bot, chat_id: int, user_id: int) -> bool: try: member = await bot.get_chat_member(chat_id, user_id) return member.status in ("administrator", "creator") except Exception: return False @router.message(Command("enable"), F.chat.type.in_({"group", "supergroup"})) async def cmd_enable(message: Message, bot: Bot): if not await is_chat_admin(bot, message.chat.id, message.from_user.id): await message.reply("Включить бота может только администратор чата.") return db = load_db() if message.chat.id not in db["groups"]: db["groups"].append(message.chat.id) save_db(db) await message.reply( "✅ Оракул включён в этом чате. Участники могут вызвать /oracle.\n" "Отключить: /disable (только админ)." ) @router.message(Command("disable"), F.chat.type.in_({"group", "supergroup"})) async def cmd_disable(message: Message, bot: Bot): if not await is_chat_admin(bot, message.chat.id, message.from_user.id): return db = load_db() if message.chat.id in db["groups"]: db["groups"].remove(message.chat.id) save_db(db) await message.reply("Оракул отключён в этом чате.") @router.message(Command("oracle"), F.chat.type.in_({"group", "supergroup"})) async def cmd_oracle(message: Message): db = load_db() if message.chat.id not in db["groups"]: return # бот молчит, пока админ не включил его — никакого спама now = time.monotonic() last = _last_oracle.get(message.chat.id, 0.0) if now - last < ORACLE_COOLDOWN_SEC: return # кулдаун: не флудим даже по запросу _last_oracle[message.chat.id] = now await message.reply(ORACLE_TEXT, reply_markup=group_kb()) @router.message(Command("broadcast"), F.chat.type == "private") async def cmd_broadcast(message: Message, bot: Bot): if message.from_user.id != ADMIN_ID: return parts = message.text.split(maxsplit=1) if len(parts) < 2: await message.answer("Формат: /broadcast <текст (HTML)>") return text = parts[1] db = load_db() sent, removed = 0, 0 for uid in list(db["subs"]): try: await bot.send_message(uid, text, reply_markup=private_kb()) sent += 1 except TelegramRetryAfter as e: await asyncio.sleep(e.retry_after + 1) try: await bot.send_message(uid, text, reply_markup=private_kb()) sent += 1 except Exception: pass except TelegramForbiddenError: db["subs"].remove(uid) # пользователь заблокировал бота — убираем removed += 1 except Exception as e: logging.warning("broadcast to %s failed: %s", uid, e) await asyncio.sleep(BROADCAST_PACE_SEC) save_db(db) await message.answer(f"Готово: доставлено {sent}, отписано (блок) {removed}.") async def main(): bot = Bot(token=BOT_TOKEN, default=DefaultBotProperties(parse_mode=ParseMode.HTML)) dp = Dispatcher() dp.include_router(router) await bot.delete_webhook(drop_pending_updates=True) logging.info("Оракул-бот запущен (polling).") await dp.start_polling(bot) if __name__ == "__main__": logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s") delay = 5 while True: # супервизор: падение сети/Telegram не убивает процесс навсегда try: asyncio.run(main()) delay = 5 except (KeyboardInterrupt, SystemExit): logging.info("Остановлено вручную.") break except Exception: logging.exception("Сбой. Перезапуск через %s сек...", delay) time.sleep(delay) delay = min(delay * 2, 300) # backoff до 5 минут, без долбёжки API
